"""Fast-forward only after unchanged local-state and path-overlap checks."""
import argparse
import hashlib
import json
import subprocess
from datetime import datetime, timezone
from pathlib import Path

PRIMARY = Path(r'C:\Users\valen\Documents\KPKNFit')
PROTECTED = [Path(r'C:\kw\int'), Path(r'C:\kw\sb'), Path(r'C:\kw\u1')]


def git(root, *args):
    return subprocess.run(['git', '-C', str(root), *args], check=True, capture_output=True).stdout


def protected_heads():
    result = {}
    for block in git(PRIMARY, 'worktree', 'list', '--porcelain').decode().strip().split('\n\n'):
        fields = dict(line.split(' ', 1) for line in block.splitlines() if ' ' in line)
        path = Path(fields['worktree'])
        if path.resolve() not in (PRIMARY.resolve(), Path(r'C:\kw\msi').resolve()):
            result[str(path)] = fields['HEAD']
    return result

def snapshot():
    status = git(PRIMARY, 'status', '--porcelain=v1', '-z', '--untracked-files=normal').decode('utf-8').split('\0')
    status = [item for item in status if item]
    assert all(item[:2] not in ('R ', ' R', 'C ', ' C') for item in status), 'Rename needs individual review'
    hashes = {}
    for entry in status:
        if entry.startswith('?? '):
            continue
        relative = entry[3:]
        assert not any(part in relative.lower() for part in ('.env', 'keystore', 'credential', 'token'))
        path = PRIMARY / relative
        assert path.resolve().is_relative_to(PRIMARY)
        hashes[relative] = hashlib.file_digest(path.open('rb'), 'sha256').hexdigest() if path.is_file() else None
    return {
        'head': git(PRIMARY, 'rev-parse', 'HEAD').decode().strip(),
        'branch': git(PRIMARY, 'branch', '--show-current').decode().strip(),
        'status': status,
        'trackedHashes': hashes,
        'staged': git(PRIMARY, 'diff', '--cached', '--name-only').decode().splitlines(),
        'protectedHeads': protected_heads(),
    }


def save(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, indent=2, ensure_ascii=False) + '\n', encoding='utf-8')


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--capture', type=Path)
    parser.add_argument('--baseline', type=Path)
    parser.add_argument('--commit')
    parser.add_argument('--receipt', type=Path)
    args = parser.parse_args()
    if args.capture:
        state = snapshot()
        assert state['branch'] == 'master' and not state['staged']
        save(args.capture, state)
        print(json.dumps({'head': state['head'], 'trackedChanges': len(state['trackedHashes']),
                          'untrackedEntries': sum(entry.startswith('?? ') for entry in state['status'])}))
        return
    assert args.baseline and args.commit and args.receipt
    before = json.loads(args.baseline.read_text(encoding='utf-8'))
    assert snapshot() == before, 'Local state changed since review; stopped'
    assert before['branch'] == 'master' and not before['staged']
    target = git(PRIMARY, 'rev-parse', args.commit + '^{commit}').decode().strip()
    git(PRIMARY, 'merge-base', '--is-ancestor', before['head'], target)
    changed = set(git(PRIMARY, 'diff', '--name-only', before['head'], target).decode().splitlines())
    assert not changed.intersection(before['trackedHashes']), 'Commit overlaps local tracked work'
    added = git(PRIMARY, 'diff', '--name-only', '--diff-filter=A', before['head'], target).decode().splitlines()
    for relative in added:
        path = PRIMARY / relative
        assert path.resolve().is_relative_to(PRIMARY)
        assert not path.exists(), 'New commit path collides with untracked or ignored work: ' + relative
        assert not any(parent.is_file() for parent in path.parents if parent.is_relative_to(PRIMARY)), 'Parent path collision: ' + relative
    result = git(PRIMARY, 'merge', '--ff-only', '--no-edit', '--no-overwrite-ignore', target)
    print(result.decode('utf-8', errors='replace'))
    after = snapshot()
    assert after['head'] == target
    assert {key: value for key, value in after.items() if key != 'head'} == {
        key: value for key, value in before.items() if key != 'head'
    }, 'Post-merge local-state drift; report without resetting'
    save(args.receipt, {
        'state': 'INTEGRATED_AND_LOCAL_WORK_PRESERVED', 'masterCommit': target,
        'previousMaster': before['head'], 'trackedChangesPreserved': len(before['trackedHashes']),
        'untrackedEntryNamesAndStatusesPreserved': sum(entry.startswith('?? ') for entry in before['status']),
        'trackedHashDrift': [], 'statusDelta': [], 'stagedPaths': [],
        'protectedHeads': after['protectedHeads'], 'pushPerformed': False,
        'completed': datetime.now(timezone.utc).isoformat(),
    })
    print('Local changes and protected worktree heads verified unchanged.')


if __name__ == '__main__':
    main()
