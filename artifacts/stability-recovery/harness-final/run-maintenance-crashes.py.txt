from pathlib import Path
import argparse
import hashlib
import json
import os
import re
import subprocess
import sys
import threading
import time
import uuid

ROOT = Path(r'C:\kw\msi')
sys.path.insert(0, str(ROOT / 'scripts/stability'))
from emulator_food_tests import ADB, LOCK, SERIAL, PACKAGE, COMPONENT, adb

parser = argparse.ArgumentParser()
parser.add_argument('--out', required=True)
parser.add_argument('--phases', nargs='+', default=['IMPORT_COPYING', 'SNAPSHOT_ROW_COPIED', 'ROLLBACK_FILE_COPYING',
    'PREPARED_SYNCED', 'PORTABLE_ROOT_REPLACED', 'DATABASE_ROW_COPIED', 'DATABASE_APPLIED', 'PREFERENCES_APPLIED',
    'COMMIT_JOURNAL_OPENED', 'COMMITTED_SYNCED', 'RECOVERY_ROOT_RESTORED', 'RECOVERY_DATABASE_APPLIED',
    'RECOVERY_SYNCED', 'REOPEN_STARTED', 'REOPEN_READBACK'])
parser.add_argument('--quota', action='store_true')
parser.add_argument('--quota-only', action='store_true')
args = parser.parse_args()
if args.quota_only:
    args.phases = []
    args.quota = True
out = ROOT / 'artifacts/stability-recovery' / args.out
assert not out.exists()
out.mkdir(parents=True)
ticket = str(uuid.uuid4())
try: LOCK.mkdir()
except FileExistsError: raise SystemExit('Another emulator owner is working; lock preserved')
owner = LOCK / 'owner.txt'
owner.write_text(f'ticket={ticket} worktree={ROOT}\n')
done = threading.Event()
def heartbeat():
    while not done.wait(15):
        try: os.utime(LOCK, None)
        except OSError: return
beat = threading.Thread(target=heartbeat, daemon=True)
beat.start()
report = dict(state='RUNNING', runId=ticket, cases=[],
    limits='Actual isolated process termination and SQLite page quota. No physically full filesystem, live Firebase, Play purchase, SAF or complete device matrix.')
runner = PACKAGE + '.test/androidx.test.runner.AndroidJUnitRunner'
cls = 'com.example.kpkn.data.backup.MaintenanceCrashRecoveryTest'

def control(case, file):
    assert re.fullmatch(r'[A-Z_]{1,80}', case) and re.fullmatch(r'[a-z.-]+', file)
    result = adb('exec-out', 'run-as', PACKAGE, 'cat', f'cache/qa-recovery-{ticket}/{case}/{file}')
    assert result.returncode == 0 and result.stdout.lstrip().startswith(b'{'), (case, file, result.stderr)
    return json.loads(result.stdout)

def invoke(case, phase, method, step=None, mode=None):
    assert re.fullmatch(r'[A-Z_]{1,80}', case) and re.fullmatch(r'[A-Z_]+', phase)
    assert adb('shell', 'am', 'force-stop', PACKAGE).returncode == 0
    assert not adb('shell', 'pidof', '-s', PACKAGE).stdout.strip()
    parameters = ['shell', 'am', 'instrument', '-w', '-r', '-e', 'class', cls + '#' + method,
        '-e', 'kpkn_run_id', ticket, '-e', 'kpkn_phase', phase, '-e', 'kpkn_case', case]
    if step: parameters += ['-e', 'kpkn_step', step]
    if mode: parameters += ['-e', 'kpkn_mode', mode]
    parameters += [runner]
    result = adb(*parameters, timeout=240)
    text = (result.stdout + result.stderr).decode(errors='replace')
    filename = f'{case}-{method}-{step or "main"}.txt'
    (out / filename).write_text(text, encoding='utf-8')
    if method == 'interrupt':
        marker = control(case, 'crash-' + (step or 'restore') + '.json')
        assert 'kpkn_crash_pid=' + str(marker['pid']) in text
        assert marker['phase'] == ('DATABASE_APPLIED' if step == 'prime' else phase)
        assert 'Process crashed' in text or 'INSTRUMENTATION_FAILED' in text, text[-1500:]
        assert str(marker['pid']) not in adb('shell', 'pidof', PACKAGE).stdout.decode().split()
        return dict(outcome='EXPECTED_PROCESS_TERMINATION_NOT_TEST_PASS', pid=marker['pid'], checkpoint=marker['phase'], marker=marker, log=filename)
    assert result.returncode == 0 and re.search(r'OK \(1 test\)', text), text[-2000:]
    pid = re.search(r'kpkn_(?:seed|verify)_pid=(\d+)', text)
    return dict(outcome='PASS', pid=int(pid.group(1)) if pid else None, log=filename)

def cleanup_case(case, profile):
    # Remove only a profile newly created and recorded by this exact fixture after its process ends.
    assert re.fullmatch(r'guest_[a-f0-9]{24}', profile)
    assert control(case, 'seed.json')['profileKey'] == profile
    assert adb('shell', 'am', 'force-stop', PACKAGE).returncode == 0
    paths = [f'{base}/profiles/{profile}' for base in ['files', 'cache', 'no_backup', 'databases']]
    assert all(re.fullmatch(r'(files|cache|no_backup|databases)/profiles/guest_[a-f0-9]{24}', p) for p in paths)
    result = adb('shell', 'run-as', PACKAGE, 'rm', '-rf', '--', *paths)
    assert result.returncode == 0

try:
    assert PACKAGE == 'com.example.kpkn.stability' and adb('get-state').stdout.strip() == b'device'
    report['api'] = adb('shell', 'getprop', 'ro.build.version.sdk').stdout.decode().strip()
    aapt = sorted(ADB.parents[1].glob('build-tools/*/aapt2.exe'))[-1]
    for kind, apk, expected in [('app', ROOT / 'android-native/app/build/outputs/apk/debug/app-debug.apk', PACKAGE),
                              ('test', ROOT / 'android-native/app/build/outputs/apk/androidTest/debug/app-debug-androidTest.apk', PACKAGE + '.test')]:
        metadata = subprocess.run([str(aapt), 'dump', 'badging', str(apk)], capture_output=True, check=True).stdout.decode()
        assert f"name='{expected}'" in metadata
        install = adb('install', '-r', '--no-incremental', str(apk), timeout=120)
        (out / f'install-{kind}.txt').write_bytes(install.stdout + install.stderr)
        assert install.returncode == 0 and b'Success' in install.stdout
        report[kind + 'ApkSha256'] = hashlib.sha256(apk.read_bytes()).hexdigest()
    assert f'instrumentation:{runner} (target={PACKAGE})' in adb('shell', 'pm', 'list', 'instrumentation').stdout.decode()
    for phase in args.phases:
        assert re.fullmatch(r'[A-Z_]+', phase)
        case = dict(phase=phase, steps=[])
        report['cases'].append(case)
        case['steps'].append(invoke(phase, phase, 'seed'))
        seed = control(phase, 'seed.json')
        case['seed'] = seed
        if phase.startswith('RECOVERY_'):
            case['steps'].append(invoke(phase, phase, 'interrupt', step='prime'))
            case['steps'].append(invoke(phase, phase, 'interrupt', step='recover'))
        else:
            case['steps'].append(invoke(phase, phase, 'interrupt', step='restore'))
        case['steps'].append(invoke(phase, phase, 'verify'))
        case['verification'] = control(phase, 'verified.json')
        pids = [step['pid'] for step in case['steps']]
        assert None not in pids and len(pids) == len(set(pids)), pids
        case['state'] = 'PASS_WITH_ACTUAL_INTERRUPTION'
        cleanup_case(phase, seed['profileKey'])
        (out / 'result.json').write_text(json.dumps(report, indent=2)+'\n')
        print(json.dumps(dict(phase=phase, state=case['state'], pids=pids)), flush=True)
    if args.quota:
        case = dict(phase='SQLITE_FULL', steps=[])
        report['cases'].append(case)
        case['steps'].append(invoke('SQLITE_FULL', 'DATABASE_APPLIED', 'seed', mode='quota'))
        seed = control('SQLITE_FULL', 'seed.json')
        case['steps'].append(invoke('SQLITE_FULL', 'DATABASE_APPLIED', 'sqliteFull'))
        case['verification'] = control('SQLITE_FULL', 'quota.json')
        assert case['verification']['sqliteFull'] is True
        case['state'] = 'PASS_SQLITE_QUOTA_FILESYSTEM_NOT_FILLED'
        cleanup_case('SQLITE_FULL', seed['profileKey'])
    launch = adb('shell', 'am', 'start', '-W', '-n', COMPONENT)
    (out / 'launch.txt').write_bytes(launch.stdout + launch.stderr)
    time.sleep(5)
    state = adb('shell', 'dumpsys', 'activity', 'activities').stdout
    resumed = b'\n'.join(line for line in state.splitlines() if b'topResumedActivity' in line or b'mResumedActivity' in line)
    (out / 'activity.txt').write_bytes(resumed)
    assert COMPONENT.encode() in resumed
    screen = adb('exec-out', 'screencap', '-p')
    assert screen.returncode == 0 and screen.stdout.startswith(b'\x89PNG\r\n\x1a\n')
    (out / 'main.png').write_bytes(screen.stdout)
    tree = adb('exec-out', 'uiautomator', 'dump', '/dev/tty')
    assert tree.returncode == 0 and b'<hierarchy' in tree.stdout
    (out / 'ui.xml').write_bytes(tree.stdout)
    pid = adb('shell', 'pidof', '-s', PACKAGE).stdout.decode().strip()
    assert pid.isdecimal()
    (out / 'logcat-main.txt').write_bytes(adb('logcat', '-d', '-v', 'threadtime', '--pid', pid).stdout)
    report['finalMainPid'] = int(pid)
    report.update(state='PASS', mainVisible=True)
except BaseException as error:
    report.update(state='FAIL_OR_INCOMPLETE', error=str(error))
    raise
finally:
    (out / 'result.json').write_text(json.dumps(report, indent=2)+'\n', encoding='utf-8')
    done.set(); beat.join(timeout=2)
    if owner.exists() and f'ticket={ticket}' in owner.read_text(): owner.unlink(); LOCK.rmdir()
    print(json.dumps(dict(state=report['state'], cases=len(report['cases']))), flush=True)
