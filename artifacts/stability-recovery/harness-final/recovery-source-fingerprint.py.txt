from pathlib import Path
from datetime import datetime, timezone
import hashlib
import json
import subprocess
import sys

root = Path(r'C:\kw\msi')
out = root / 'artifacts/stability-recovery'
target = out / sys.argv[1]
assert not target.exists()
baseline = json.loads((root / 'artifacts/stability-integrated-sessions/tested-source-fingerprint-final.json').read_text())
files = {r['path'] for r in baseline['files']}
changed = subprocess.run(['git', '-C', str(root), 'diff', '--name-only', '--', 'android-native/app/src'],
    capture_output=True, check=True).stdout.decode().splitlines()
added = subprocess.run(['git', '-C', str(root), 'ls-files', '--others', '--exclude-standard', '--', 'android-native/app/src'],
    capture_output=True, check=True).stdout.decode().splitlines()
files.update(changed); files.update(added)
result = dict(baselineCommit='11aff1778c2b7a268f2f77f2989ff71622a2cad9', recorded=datetime.now(timezone.utc).isoformat(),
    changedSources=changed, addedSources=added, files=[])
for relative in sorted(files):
    p = root / relative
    assert p.resolve().is_relative_to(root) and p.is_file()
    assert p.name not in ('.env', 'google-services.json', 'local.properties') and p.suffix not in ('.jks','.keystore','.p12')
    result['files'].append(dict(path=relative, sha256=hashlib.sha256(p.read_bytes()).hexdigest()))
result['fileCount'] = len(result['files'])
target.write_text(json.dumps(result, indent=2)+'\n', encoding='utf-8')
print(json.dumps(dict(sourceFiles=len(files), changed=len(changed), added=len(added))))
